    }

                    credentialsId: 'dokdokhae_token_username_with_password',
                    url: 'https://lab.ssafy.com/s13-bigdata-recom-sub1/S13P21A104.git'
            }
        }

        stage('Detect Changed Dirs') {
            steps {
                script {
                    def changedFiles = sh(
                        script: "git diff --name-only HEAD~1 HEAD",
                        returnStdout: true
                    ).trim().split("\n")

                    changedFiles = changedFiles.findAll { it?.trim() }
                    def changedDirsList = changedFiles.collect { it.split('/')[0] }.unique()
                    env.CHANGED_DIRS = changedDirsList.join(",")

                    echo "Changed files: ${changedFiles}"
                    echo "Changed dirs (list): ${changedDirsList}"
                    echo "Changed dirs (env): ${env.CHANGED_DIRS}"

                    env.CURRENT_BRANCH = sh(
                        script: "git rev-parse --abbrev-ref HEAD",
                        returnStdout: true
                    ).trim()
                    echo ">>> CURRENT_BRANCH = ${env.CURRENT_BRANCH}"
                }
            }
        }

        stage('CI - be-ai') {
          when { expression { env.CHANGED_DIRS.split(",").contains('be-ai') } }
          steps {
            dir('be-ai') {
              withCredentials([file(credentialsId: 'AI_APP_SECRET', variable: 'AI_SECRET_FILE')]) {

                  rm -f ./.env ./.env.prod

                echo "$DOCKER_TOKEN" | docker login -u "$DOCKER_USER" --password-stdin

                docker push $DOCKERHUB_REPO/be-ai:latest
              '''
            }
          }
        }


        stage('CI - be-biz') {
            when { expression { env.CHANGED_DIRS.split(",").contains('be-biz') } }
            steps {
                dir('be-biz') {
                    withCredentials([
                        file(credentialsId: 'BEBIZ_APP_SECRET', variable: 'BEBIZ_SECRET_FILE'),

                          cp "$BEBIZ_SECRET_FILE" ./src/main/resources/application-secret.yml
